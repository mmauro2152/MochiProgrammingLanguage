program fibProgram;
var n: int; i: int; a: int; b: int; temp: int;

main {
    n = 10;          

    a = 0;           
    b = 1;           

    print(a);        
    print(b);        

    i = 2;
    while (i < n) do {
        temp = a + b;
        print(temp);

        a = b;
        b = temp;

        i = i + 1;
    };
}
end
