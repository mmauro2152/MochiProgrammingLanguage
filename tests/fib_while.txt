program fibProgram;
var n: int; i: int; a: int; b: int; temp: int;

main {
    n = 10;          

    a = 0;           
    b = 1; 
    i = 0;          

    print(i++, ": ", a);        
    print(i++, ": ", b);        

    while (i < n) do {
        temp = a + b;
        print(i++, ": ", temp);

        a = b;
        b = temp;
    };
}
end
