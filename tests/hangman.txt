program hangman;
var in, word, user_word: string; i, hp: int;

void print_hearts() [
var temp, h: string;
{
    h = "<3";
    temp = "";

    for (i = 0; i < hp; i++) {
        temp = temp + h + " ";
    };

    print(temp);
}];

void reserve_user_word() [{
    user_word = "";
    for (i = 0; i < len(word); i++) {
        user_word = user_word + "_";
    };
}];

bool is_complete() [{
    return user_word == word;
}];

void fill(c: string) [
var temp: string; correct: bool;
{
    temp = "";
    correct = false;

    for (i = 0; i < len(word); i++) {
        if (word[i] == upper(c) || word[i] == lower(c)) {
            temp = temp + word[i];
            correct = true;
        } else {
            temp = temp + user_word[i];
        };
    };

    if (!correct) {
        hp--;
    };

    user_word = temp;
}];

void clear_console() [
{
    for (i = 0; i < 100; i++) {
        print("");
    };
}];

main {
    clear_console();

    word = "Compilador";
    hp = 10;
    reserve_user_word();

    print_hearts();
    print("adivina la palabra: ", user_word);
    print("");

    do {
        input(in);

        if (len(in) > 1) {
            clear_console();
            print_hearts();
            print("adivina la palabra: ", user_word);
            print("¡escribe solo una letra!");
            print("");
        } else {
            fill(in);

            clear_console();
            print_hearts();
            print("adivina la palabra: ", user_word);
            print("");
        };

    } while (!is_complete() && hp > 0);

    if (hp == 0) {
        clear_console();
        print("¡Perdiste!");
        print("La palabra era: ", word);
    } else {
        print("¡Felicidades! Adivinaste la palabra");
    };
}
end
