program demo;
var A, B, C, D, E, F, G, H, J, K: float;
main {
    A = B + C * (D-E/F) * H;
    B = E - F;
    while (!(A*B >= D*E/(G*H))) do {
        H = J * K + -B;
        if (B < H){
            B = H + J;
            while (B > A + C) do {
                A = A+B*C;
                F = D-E;
            };
        }
        else{
            while (A-D < C+-B) do {
                A = -A + -B;
                C = B-D;
            };
        };
    };
}
end
